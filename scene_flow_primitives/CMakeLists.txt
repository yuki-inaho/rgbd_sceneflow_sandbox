cmake_minimum_required(VERSION 3.20)
project(scene_flow_refinement LANGUAGES CXX)
add_library(_native SHARED cpp/bindings_c.cpp cpp/matcher.cpp cpp/solver.cpp cpp/renderer.cpp cpp/rigid.cpp)
target_compile_features(_native PRIVATE cxx_std_17)
set_target_properties(_native PROPERTIES PREFIX "" OUTPUT_NAME "_native")
if(MSVC)
  target_compile_options(_native PRIVATE /W4)
else()
  target_compile_options(_native PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS _native LIBRARY DESTINATION scene_flow_refinement RUNTIME DESTINATION scene_flow_refinement)
